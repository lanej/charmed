Foundations

Themes, cell geometry, focus and motion are shared across every component. Choose semantic roles rather than embedding color decisions in application widgets. Layout parents allocate rectangles in terminal columns and rows; focus scopes route keys to the active child.

The default saved frame is the truecolor reference. Separate tabs show narrow, 256-color, 16-color and monochrome presentations; fallback color limitations do not define the reference palette. Run with --theme light, --columns 40 or --ascii --profile none to inspect those presentations in a terminal. The HTML renderer reads completed static cells, preserves RGB and escapes text; it does not emulate cursor movement or other live terminal effects.

Colors and emphasis

Foreground and muted text are neutral. Accent identifies interaction; success, information, warning and danger identify semantic meaning. Categorical series use a separate palette and never alias those interface or status tokens. Series order remains stable across views; labels and matching legend keys retain identity when fallback colors converge.

Bright, normal and dim emphasis preserve the label, glyph and cell allocation. Dim uses explicit shades of each semantic hue instead of blending every tone toward gray or relying on the terminal's faint attribute. The built-in dim text tokens meet a 4.5:1 contrast floor against their theme background; native fonts, color profiles and terminal settings still affect appearance. Monochrome keeps labels, glyphs and bright weight; normal and dim may coincide without color.

Glyphs and spacing

theme.Glyph names status, selection, expansion, navigation, loading and unavailable marks. Theme.Glyph selects a one-cell Unicode or ASCII representation. A glyph supplements a semantic text label; UNKNOWN and OFF remain different states. Animated spinners retain native frame sequences, with a named static loading mark for reduced motion. Application-specific chart markers remain caller-owned.

theme.InlinePadding reserves one horizontal cell on each side of controls, pills and panel content. Theme.Control and Theme.Focus allocate the same space. Status always composes its glyph, one separating cell and its label inside the same padded pill; emphasis changes color and weight without changing geometry. Monochrome focus retains inverse highlighting. Theme.Selection uses a quieter selection surface so nested table text and semantic pills remain readable; tables reuse their existing cell insets instead of adding a second focus inset.

Panel dimensions include borders and content gutters. Allocate children with layout.PanelContentSize; chart.PanelPlotSize also reserves an optional legend row. At widths below five cells, panels reduce the gutter to preserve their allocation. Consumers previously allocating only two border cells must reserve the new content gutters as well. Theme literals without explicit dim tokens retain their supplied normal tone until the caller provides dim shades; literals without a selection surface use bold and underline to retain a visible selection without assuming fill contrast.

func layoutStory(e Environment) (runtime.Scene, error) {
	return static(func(size layout.Size) string {
		t := e.Config.Palette()
		left, right, split := layout.Split(size.Width, 1, 20, 30, .4)
		height := max(3, size.Height-4)
		a := layout.Panel("Navigation", "Cell-based dimensions\nBorders included in size", layout.Size{Width: left, Height: height}, t, true)
		b := layout.Panel("Content", "The right pane adapts to available space.\nNarrow terminals stack the same panels.", layout.Size{Width: right, Height: height}, t, false)
		if split {
			return title("Responsive layout", e) + layout.Row(a, " ", b)
		}
		height = max(3, height/2)
		return title("Stacked layout", e) + layout.Column(layout.Panel("Navigation", "Same content", layout.Size{Width: left, Height: height}, t, true), layout.Panel("Content", "Same state", layout.Size{Width: right, Height: height}, t, false))
	}), nil
}

func motionStory(e Environment) (runtime.Scene, error) {
	m := feedback.NewMotion(e.Config.Palette())
	return &scene{update: func(msg tea.Msg) tea.Cmd {
		if _, ok := msg.(runtime.StepMsg); ok {
			m.Step(e.Config.Amount)
			emit(e, map[string]any{"type": "motion", "position": m.Position})
		}
		return nil
	}, view: func(layout.Size) string {
		return title("Spring motion", e) + fmt.Sprintf("Position %.2f  /  Target %.0f\n\nVelocity %.2f", m.Position, e.Config.Amount, m.Velocity) + footer("Ctrl+N steps the spring. Reduced motion resolves immediately.", e)
	}}, nil
}

func themeStory(e Environment) (runtime.Scene, error) {
	return static(func(size layout.Size) string {
		t := e.Config.Palette()

		rows := []string{title("Semantic tokens / normal and dim", e), t.Text().Render("Primary content"), t.Subtle().Render("Supporting information"), t.Heading().Render("Accent / interaction"), ""}
		for _, tone := range []text.Tone{text.Neutral, text.Success, text.Info, text.Warning, text.Danger} {
			rows = append(rows, text.ToneStyle(tone, text.Normal, t).Render(fmt.Sprintf("%-9s", tone))+"  "+text.ToneStyle(tone, text.Dim, t).Render("dim "+string(tone)))
		}

		rows = append(rows, "", "Categorical data / independent palette")

		var series []string
		for i, color := range t.Series {
			series = append(series, t.Text().Foreground(color).Render(fmt.Sprintf("%d ━━", i+1)))
		}

		rows = append(rows, strings.Join(series, "  "))

		return strings.Join(rows, "\n")
	}), nil
}

func glyphStory(e Environment) (runtime.Scene, error) {
	return static(func(layout.Size) string {
		t := e.Config.Palette()
		fallback := t
		fallback.ASCII = true
		rows := []string{"Meaning     U A     Meaning     U A", ""}

		var cells []string
		for _, glyph := range []theme.Glyph{theme.GlyphOK, theme.GlyphWarning, theme.GlyphError, theme.GlyphUnknown, theme.GlyphOff, theme.GlyphInfo, theme.GlyphSelected, theme.GlyphUnselected, theme.GlyphExpanded, theme.GlyphCollapsed, theme.GlyphUp, theme.GlyphDown, theme.GlyphLeft, theme.GlyphRight, theme.GlyphLoading, theme.GlyphUnavailable} {
			cells = append(cells, layout.Fit(fmt.Sprintf("%-11s %s %s", glyph, t.Glyph(glyph), fallback.Glyph(glyph)), layout.Size{Width: 18, Height: 1}))
		}

		for i := 0; i < len(cells); i += 2 {
			rows = append(rows, cells[i]+"  "+cells[i+1])
		}

		return strings.Join(rows, "\n")
	}), nil
}

func focusStory(e Environment) (runtime.Scene, error) {
	return static(func(size layout.Size) string {
		t := e.Config.Palette()

		rows := []string{"Stable control insets / 1 cell per side", "", "Normal  |" + t.Control().Render("Value") + "|", "Focused |" + t.Focus().Render("Value") + "|", "Row     |" + t.Selection().Render("Value") + "|", ""}
		for _, emphasis := range []text.Emphasis{text.Bright, text.Normal, text.Dim} {
			rows = append(rows, fmt.Sprintf("%-7s |", emphasis)+text.Pill("Value", text.Info, t, emphasis)+"|")
		}

		rows = append(rows, "", layout.Panel("Content gutter", "Text starts one cell from the border.", layout.Size{Width: size.Width, Height: 3}, t, true))

		return strings.Join(rows, "\n")
	}), nil
}
