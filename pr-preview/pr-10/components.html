package main

import (
	"fmt"
	"math"
	"time"

	"github.com/NimbleMarkets/ntcharts/v2/linechart/timeserieslinechart"
	"github.com/lanej/charmed/chart"
	"github.com/lanej/charmed/layout"
	"github.com/lanej/charmed/theme"
)

func main() {
	palette := theme.New(theme.Dark)
	size := layout.Size{Width: 80, Height: 18}
	series := []chart.TimeSeriesData{{ID: "requests", Label: "Requests", Unit: "req/min"}}
	start := time.Date(2026, 1, 1, 12, 0, 0, 0, time.UTC)
	for i := range 32 {
		series[0].Points = append(series[0].Points, timeserieslinechart.TimePoint{
			Time:  start.Add(time.Duration(i) * time.Minute),
			Value: 800 + 120*math.Sin(float64(i)*0.3),
		})
	}
	plot := chart.NewTimeSeries(series, chart.PanelPlotSize(size), palette)
	fmt.Println(chart.Panel("Request throughput", plot, series, size, palette, false))
}

func chartStory(kind string, e Environment) (runtime.Scene, error) {
	t := e.Config.Palette()
	step := 0
	currentSize := e.Config.Size()
	var render func() string
	var update func(tea.Msg) tea.Cmd
	var rebuild func(layout.Size)
	rebuild = func(size layout.Size) {
		currentSize = size
		plotSize := layout.Size{Width: size.Width, Height: max(1, size.Height-6)}
		update = nil
		switch kind {
		case "sparkline":
			values := sampleValues(48, step, e.Config.Amount)
			plot := chart.NewSparkline(values, layout.Size{Width: size.Width, Height: max(1, min(5, plotSize.Height))}, t)
			render = func() string { return plot.View() + fmt.Sprintf("\n\nLatest %.1f req/min", values[len(values)-1]) }
		case "timeseries":
			data := sampleSeries(e.Config.Series, step, e.Config.Amount)
			plot := chart.NewTimeSeries(data, plotSize, t)
			plot.Focus()
			render = func() string {
				view := plot.View()
				if e.Config.Legend {
					view += "\n" + chart.Legend(data, t)
				}
				return view
			}
			update = func(msg tea.Msg) tea.Cmd { var cmd tea.Cmd; plot, cmd = plot.Update(msg); return cmd }
		case "line", "scatter":
			plot := chart.NewLine(plotSize, 0, 31, 0, 120, t)
			plot.Focus()
			draw := func() {
				plot.Clear()
				plot.DrawXYAxisAndLabel()
				for s := range e.Config.Series {
					values := sampleValues(32, step+s*3, e.Config.Amount-float64(s*10))
					style := lipgloss.NewStyle().Foreground(t.SeriesColor(s))
					for i, v := range values {
						p := canvas.Float64Point{X: float64(i), Y: v}
						if kind == "scatter" {
							plot.DrawRuneWithStyle(p, '●', style)
						} else if i > 0 {
							plot.DrawBrailleLineWithStyle(canvas.Float64Point{X: float64(i - 1), Y: values[i-1]}, p, style)
						}
					}
				}
			}
			draw()
			render = plot.View
			update = func(msg tea.Msg) tea.Cmd { var cmd tea.Cmd; plot, cmd = plot.Update(msg); draw(); return cmd }
		case "bars":
			opts := []barchart.Option{}
			if e.Config.Horizontal {
				opts = append(opts, barchart.WithHorizontalBars())
			}
			plot := chart.NewBars(plotSize, t, opts...)
			for i, name := range []string{"API", "Events", "Jobs", "Storage", "Schedule"} {
				values := []barchart.BarValue{}
				for s := range e.Config.Series {
					values = append(values, barchart.BarValue{Name: fmt.Sprintf("Series %d", s+1), Value: 20 + float64((i+s+step)*13%50), Style: lipgloss.NewStyle().Foreground(t.SeriesColor(s))})
				}
				plot.Push(barchart.BarData{Label: name, Values: values})
			}
			plot.Draw()
			render = plot.View
		case "heatmap":
			plot := chart.NewHeatmap(plotSize, t, heatmap.WithValueRange(0, 100), heatmap.WithCellSize(1, 1))
			plot.SetXYRange(0, 12, 0, 8)
			plot.SetViewXYRange(0, 12, 0, 8)
			for x := range 12 {
				for y := range 8 {
					plot.Push(heatmap.NewHeatPoint(float64(x), float64(y), 50+45*math.Sin(float64(x+y+step)*.4)))
				}
			}
			plot.Draw()
			render = plot.View
		case "stream":
			plot := chart.NewStream(plotSize, t, streamlinechart.WithYRange(0, 120))
			for _, v := range sampleValues(max(32, size.Width), step, e.Config.Amount) {
				plot.Push(v)
			}
			plot.DrawAll()
			render = plot.View
		case "wave":
			plot := chart.NewWave(plotSize, t, wavelinechart.WithXYRange(0, 31, 0, 120))
			for i, v := range sampleValues(32, step, e.Config.Amount) {
				plot.PlotDataSet("observed", canvas.Float64Point{X: float64(i), Y: v})
			}
			plot.DrawAll()
			render = plot.View
		case "ohlc":
			data := sampleSeries(4, step, e.Config.Amount)
			names := []string{"open", "high", "low", "close"}
			for i := range data {
				data[i].ID = names[i]
				data[i].Label = names[i]
				for p := range data[i].Points {
					base := sampleValues(32, step, e.Config.Amount)[p]
					offset := []float64{0, 9, -7, 4}[i]
					data[i].Points[p].Value = base + offset
				}
			}
			plot := chart.NewTimeSeries(data, plotSize, t)
			plot.Bubble.Clear()
			plot.Bubble.DrawCandle("open", "high", "low", "close", t.Text().Foreground(t.Success), t.Text().Foreground(t.Danger))
			render = plot.Bubble.View
		case "canvas":
			plot := chart.NewCanvas(plotSize, t)
			for x := range max(0, size.Width) {
				for y := range max(0, plotSize.Height) {
					if (x+y+step)%7 == 0 {
						plot.SetRuneWithStyle(canvas.Point{X: x, Y: y}, '·', t.Subtle())
					}
				}
			}
			plot.SetStringWithStyle(canvas.Point{X: 2, Y: 2}, "CHARMED / ntcharts canvas", t.Heading())
			render = plot.View
		}
	}
	tick := func() tea.Cmd { return tea.Tick(time.Second, func(time.Time) tea.Msg { return chartPulse{} }) }
	s := &scene{init: func() tea.Cmd {
		if e.Config.Live && !t.ReducedMotion {
			return tick()
		}
		return nil
	}, resize: rebuild, update: func(msg tea.Msg) tea.Cmd {
		_, manual := msg.(runtime.StepMsg)
		_, pulse := msg.(chartPulse)
		if manual || pulse {
			step++
			rebuild(currentSize)
			emit(e, map[string]any{"type": "chart-step", "step": step})
			if pulse && e.Config.Live && !t.ReducedMotion {
				return tick()
			}
			return nil
		}
		if update != nil {
			return update(msg)
		}
		return nil
	}, view: func(size layout.Size) string {
		if stateMessage(e) != "" {
			return title("Chart / "+kind, e) + stateMessage(e)
		}
		if size.Width < 20 || size.Height < 12 {
			return title(kind, e) + "Increase terminal size to see the chart."
		}
		if t.ASCII {
			return title(kind+" / text fallback", e) + fmt.Sprintf("%d series / latest %.1f\n\nASCII mode presents values without chart glyphs.", e.Config.Series, e.Config.Amount)
		}
		return title("Chart / "+kind, e) + render() + footer("Ctrl+N advances fixture data / arrows pan supported plots", e)
	}}
	rebuild(e.Config.Size())
	return s, nil
}

// IntervalBand is a one-row ntcharts canvas primitive. Density encodes nested
// intervals independently of color; a selected observation gets a distinct mark.
// Invalid geometry produces an explicit fallback instead of a misleading band.
func IntervalBand(o BandOptions, width int, t theme.Theme) string {
	if width <= 0 {
		return ""
	}
	if !validBand(o) {
		return layout.Fit("Invalid interval", layout.Size{Width: width, Height: 1})
	}

	plot := NewCanvas(layout.Size{Width: width, Height: 1}, t)
	style := bandStyle(o, t)
	for x := range width {
		plot.SetRuneWithStyle(canvas.Point{X: x}, '·', t.Subtle())
		if t.ASCII {
			plot.SetRuneWithStyle(canvas.Point{X: x}, '.', t.Subtle())
		}
	}
	project := func(v float64) int {
		return int(math.Round((max(o.Min, min(o.Max, v)) - o.Min) / (o.Max - o.Min) * float64(width-1)))
	}
	intervals := append([]Interval(nil), o.Intervals...)
	sort.SliceStable(intervals, func(i, j int) bool { return intervals[i].High-intervals[i].Low > intervals[j].High-intervals[j].Low })
	glyphs := []rune{'░', '▒', '▓'}
	if t.ASCII {
		glyphs = []rune{'.', '=', '#'}
	}
	for level, i := range intervals {
		if i.High < o.Min || i.Low > o.Max {
			continue
		}
		for x := project(i.Low); x <= project(i.High); x++ {
			plot.SetRuneWithStyle(canvas.Point{X: x}, glyphs[min(level+max(0, 3-len(intervals)), len(glyphs)-1)], style)
		}
	}
	if o.Center >= o.Min && o.Center <= o.Max {
		plot.SetRuneWithStyle(canvas.Point{X: project(o.Center)}, '|', style.Bold(true))
	}
	for _, marker := range o.Markers {
		if finite(marker.Value) && marker.Value >= o.Min && marker.Value <= o.Max {
			plot.SetRuneWithStyle(canvas.Point{X: project(marker.Value)}, marker.Glyph, marker.Style)
		}
	}
	if o.Selected != nil && finite(*o.Selected) && *o.Selected >= o.Min && *o.Selected <= o.Max {
		plot.SetRuneWithStyle(canvas.Point{X: project(*o.Selected)}, 'X', selectedStyle(t))
	}
	return layout.Fit(plot.View(), layout.Size{Width: width, Height: 1})
}

type DistributionSeries struct {

	HasSpread    bool
	Selected     *float64

// DistributionComparison composes mean/spread labels, interval atoms, shared
// caller-supplied boundary markers and a common axis. Values use the caller's unit.
type DistributionComparison struct {
	Series           []DistributionSeries
	Sigma            int
	Min, Max         float64
	Unit, SpreadUnit string
	Markers          []BandMarker
	// Compact puts a direct series label and mean beside each band.
	Compact    bool
	ShowLegend bool
}

func (d DistributionComparison) View(width int, t theme.Theme) string {
	low, high := math.Inf(1), math.Inf(-1)
	for _, s := range d.Series {
		if s.Members > 0 {
			spread := 0.0
			if s.HasSpread {
				spread = 3 * s.StdDev
			}
			low = min(low, s.Mean-spread)
			high = max(high, s.Mean+spread)
		}
		if s.Selected != nil && finite(*s.Selected) {
			low = min(low, *s.Selected)
			high = max(high, *s.Selected)
		}
	}
	for _, m := range d.Markers {
		if finite(m.Value) && m.Value >= d.Min && m.Value <= d.Max {
			low = min(low, m.Value)
			high = max(high, m.Value)
		}
	}
	if !math.IsInf(low, 0) {
		pad := max((high-low)*0.07, (d.Max-d.Min)*0.00001)
		low = max(d.Min, low-pad)
		high = min(d.Max, high+pad)
	}
	var lines []string
	var legend []legendEntry
	prefixWidth := 0
	if d.Compact {
		prefixWidth = min(29, max(8, width/3))
	}
	for i, s := range d.Series {
		style := t.Text().Foreground(t.SeriesColor(i))
		if s.Members == 0 {
			if d.Compact {
				lines = append(lines, style.Render(ansi.Truncate(s.Label, max(1, prefixWidth-5), "~")+" n/a"))
			} else {
				lines = append(lines, style.Render(s.Label+" host mean n/a"), "Insufficient member samples")
			}
			continue
		}
		spread := "n/a"
		if s.HasSpread {
			spread = fmt.Sprintf("%.3f%s", s.StdDev, d.SpreadUnit)
		}
		if !d.Compact {
			lines = append(lines, style.Render(fmt.Sprintf("%s mean %.3f%s s=%s n=%d", s.Label, s.Mean, d.Unit, spread, s.Members)))
		}
		opts := BandOptions{Min: low, Max: high, Center: s.Mean, Selected: s.Selected, Series: i, Markers: d.Markers}
		if s.HasSpread {
			for k := max(1, min(3, d.Sigma)); k >= 1; k-- {
				opts.Intervals = append(opts.Intervals, Interval{Low: s.Mean - float64(k)*s.StdDev, High: s.Mean + float64(k)*s.StdDev})
			}
		}
		prefix := ""
		if d.Compact {
			value := fmt.Sprintf("%.3f%s", s.Mean, d.Unit)
			labelWidth := max(1, prefixWidth-len(value)-2)
			label := ansi.Truncate(s.Label, labelWidth, "~")
			prefix = style.Render(layout.Fit(label+strings.Repeat(" ", max(1, labelWidth-lipgloss.Width(label)+1))+value,
				layout.Size{Width: prefixWidth, Height: 1}))
		}
		lines = append(lines, prefix+IntervalBand(opts, max(1, width-prefixWidth), t))
		if d.ShowLegend {
			legend = append(legend, opts.legendEntries(s.Label+" ", t)...)
		}
	}
	if math.IsInf(low, 0) {
		lines = append(lines, "No eligible host spread")
	} else {
		left, right := fmt.Sprintf("%.3f%s", low, d.Unit), fmt.Sprintf("%.3f%s", high, d.Unit)
		lines = append(lines, strings.Repeat(" ", prefixWidth)+left+strings.Repeat(" ", max(1, width-prefixWidth-len(left)-len(right)))+right)
	}
	if d.ShowLegend {
		if key := renderLegend(legend, width, t); key != "" {
			lines = append(lines, key)
		}
	}
	return strings.Join(lines, "\n")
}

				opts.Intervals = append(opts.Intervals, Interval{Low: s.Mean - float64(k)*s.StdDev, High: s.Mean + float64(k)*s.StdDev})
			}
		}
		prefix := ""
		if d.Compact {
			value := fmt.Sprintf("%.3f%s", s.Mean, d.Unit)
			labelWidth := max(1, prefixWidth-len(value)-2)
			label := ansi.Truncate(s.Label, labelWidth, "~")
			prefix = style.Render(layout.Fit(label+strings.Repeat(" ", max(1, labelWidth-lipgloss.Width(label)+1))+value,
				layout.Size{Width: prefixWidth, Height: 1}))
		}
		lines = append(lines, prefix+IntervalBand(opts, max(1, width-prefixWidth), t))
		if d.ShowLegend {
			legend = append(legend, opts.legendEntries(s.Label+" ", t)...)
		}
	}
	if math.IsInf(low, 0) {
		lines = append(lines, "No eligible host spread")
	} else {
		left, right := fmt.Sprintf("%.3f%s", low, d.Unit), fmt.Sprintf("%.3f%s", high, d.Unit)
		lines = append(lines, strings.Repeat(" ", prefixWidth)+left+strings.Repeat(" ", max(1, width-prefixWidth-len(left)-len(right)))+right)
	}
	if d.ShowLegend {
		if key := renderLegend(legend, width, t); key != "" {
			lines = append(lines, key)
		}
	}
	return strings.Join(lines, "\n")
}

type DeltaProperty struct {
	Value, Extent float64

func (d DeltaProperty) View(width int, t theme.Theme) string {
	if !d.Available || !finite(d.Value) || !finite(d.Extent) || d.Extent <= 0 {
		return layout.Fit("n/a", layout.Size{Width: width, Height: 1})
	}
	number := text.SignedNumber{Value: d.Value, Available: true, Precision: 3, Unit: "pp", Tones: d.Tones}
	label := number.Label()
	labelWidth := max(10, len(label))
	barWidth := width - labelWidth - 1
	if barWidth%2 == 0 {
		barWidth--
	}
	if barWidth < 3 {
		return layout.Fit(number.View(t), layout.Size{Width: width, Height: 1})
	}
	style := d.Style
	bar := IntervalBand(BandOptions{Min: -d.Extent, Max: d.Extent, Center: 0, Intervals: []Interval{{Low: min(0, d.Value), High: max(0, d.Value)}}, Style: &style}, barWidth, t)
	return layout.Fit(bar+" "+strings.Repeat(" ", max(0, labelWidth-len(label)))+number.View(t), layout.Size{Width: width, Height: 1})
}

// SymmetricExtent rounds an observed magnitude up to a readable 1/2/5 domain.
func SymmetricExtent(magnitude float64) float64 {
	if !finite(magnitude) || magnitude <= 0 {
		return 1
	}
	power := math.Pow(10, math.Floor(math.Log10(magnitude)))
	for _, step := range []float64{1, 2, 5, 10} {
		if magnitude <= step*power {
			return step * power
		}
	}
	return magnitude
}

// SignTones is application policy: positive can mean improvement or regression.
// Unspecified positive/negative tones default to success/danger respectively.
type SignTones struct{ Positive, Negative, Zero Tone }

// SignedNumber keeps exact sign and units when color is unavailable. Zero defaults to
// neutral; missing and non-finite observations are unavailable, never zero.
type SignedNumber struct {
	Value     float64
	Available bool
	Precision int
	Unit      string
	Tones     SignTones
	Emphasis  Emphasis
}

func (n SignedNumber) valid() bool {
	return n.Available && !math.IsNaN(n.Value) && !math.IsInf(n.Value, 0)
}
func (n SignedNumber) Label() string {
	if !n.valid() {
		return "n/a"
	}
	value := n.Value
	if value == 0 {
		value = 0
	} // Canonicalize IEEE negative zero.
	precision := max(0, min(9, n.Precision))
	if value != 0 && math.Abs(value) < .5*math.Pow10(-precision) {
		return fmt.Sprintf("%+.3g%s", value, n.Unit)
	}
	return fmt.Sprintf("%+.*f%s", precision, value, n.Unit)
}
func (n SignedNumber) View(t theme.Theme) string {
	tone := n.Tones.Zero
	if !n.valid() {
		tone = Neutral
	}
	if n.valid() && n.Value > 0 {
		tone = n.Tones.Positive
		if tone == "" {
			tone = Success
		}
	} else if n.valid() && n.Value < 0 {
		tone = n.Tones.Negative
		if tone == "" {
			tone = Danger
		}
	}
	return ToneStyle(tone, n.Emphasis, t).Render(n.Label())
}

func Pill(value string, tone Tone, t theme.Theme, emphasis ...Emphasis) string {
	level := Bright // Preserve existing filled pills for three-argument calls.
	if len(emphasis) > 0 {
		level = emphasis[0]
	}
	if t.ASCII {
		return ToneStyle(tone, level, t).Render("[" + value + "]")
	}
	fill := toneColor(tone, t)
	style := ToneStyle(tone, level, t).Padding(0, 1)
	switch level {
	case Bright:
		style = style.Foreground(t.Background).Background(fill)
	case Dim:
		style = style.Background(blend(fill, t.Background, .04))
	default:
		style = style.Background(blend(fill, t.Background, .15))
	}
	return style.Render(value)
}
